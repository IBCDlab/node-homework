SELECT line_items.order_id, SUM(line_items.quantity * products.price) AS total_price FROM line_items JOIN products ON line_items.product_id = products.product_id GROUP BY line_items.order_id ORDER BY line_items.order_id LIMIT 5;
SELECT customers.customer_name, AVG(order_totals.total_price) AS average_order_price FROM (SELECT orders.customer_id, orders.order_id, SUM(line_items.quantity * products.price) AS total_price FROM orders JOIN line_items ON orders.order_id = line_items.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY orders.customer_id, orders.order_id) AS order_totals JOIN customers ON order_totals.customer_id = customers.customer_id GROUP BY customers.customer_id, customers.customer_name ORDER BY customers.customer_name;
